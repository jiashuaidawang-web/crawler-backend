# ========== JobController (/api/job) ==========

# 1. stock_daily —— 按页拆任务(每页100条)
curl -X POST "http://localhost:8081/api/job/seed-stock-daily" -d "date=2026-08-13&source=1"

# 2. board_basic —— 地域/行业/概念板块基础维表
curl -X POST "http://localhost:8081/api/job/seed-board-basic" -d "date=2026-08-13&source=1"

# 3. board_daily —— 地域/行业/概念板块日线
curl -X POST "http://localhost:8081/api/job/seed-board-daily" -d "date=2026-08-13&source=1"

# 4. limit_up_pool / limit_down_pool / zhaban_pool / strong_pool / cixin_pool(5个池子一起)
curl -X POST "http://localhost:8081/api/job/seed-pools-only" -d "date=2026-08-13&source=1"

# 5. stock_board_rel —— 板块-个股关联(依赖 board_basic)
curl -X POST "http://localhost:8081/api/job/sync-board-relations" -d "date=2026-08-13&source=1"

# 6. 指数
curl -X POST "http://localhost:8081/api/crawl/seed-index-daily?tradeDate=2026-08-13&source=1"

# 7.  跑主表
curl -X POST "http://localhost:8081/api/crawl/seed-dragon-tiger" -H "Content-Type: application/json" -d '{"tradeDate":"2026-08-13","source":1}'

# 8. 等主表跑完，跑席位明细
curl -X POST "http://localhost:8081/api/crawl/chain-seat-details" -H "Content-Type: application/json" -d '{"tradeDate":"2026-08-13"}'

# 9. 主力流入
curl -X POST "http://localhost:8081/api/crawl/seed" -H "Content-Type: application/json" -d "{\"taskType\":\"MAIN_FUND_FLOW\",\"tradeDate\":\"2026-08-13\",\"source\":1}"
